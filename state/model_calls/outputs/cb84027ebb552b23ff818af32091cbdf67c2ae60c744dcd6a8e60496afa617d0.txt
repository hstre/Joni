def solve(payload: dict) -> dict:
    def parse_date(s):
        if '-' in s:
            parts = s.split('-')
            if len(parts) != 3:
                raise ValueError
            return (int(parts[0]), int(parts[1]), int(parts[2]))
        elif '.' in s:
            parts = s.split('.')
            if len(parts) != 3:
                raise ValueError
            d, m, y = int(parts[0]), int(parts[1]), int(parts[2])
            return (y, m, d)
        else:
            raise ValueError("Unknown date format")

    a_date = parse_date(payload['a'])
    b_date = parse_date(payload['b'])

    if a_date < b_date:
        label = 'before'
    elif a_date > b_date:
        label = 'after'
    else:
        label = 'same'

    return {'label': label}
