def solve(payload: dict) -> dict:
    def parse_date(s):
        if '-' in s:
            parts = s.split('-')
            year, month, day = int(parts[0]), int(parts[1]), int(parts[2])
        else:
            parts = s.split('.')
            day, month, year = int(parts[0]), int(parts[1]), int(parts[2])
        return (year, month, day)

    a_date = parse_date(payload['a'])
    b_date = parse_date(payload['b'])

    if a_date < b_date:
        label = 'before'
    elif a_date > b_date:
        label = 'after'
    else:
        label = 'same'

    return {'label': label}
