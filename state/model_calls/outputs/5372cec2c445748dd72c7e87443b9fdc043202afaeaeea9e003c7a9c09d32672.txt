def solve(payload: dict) -> dict:
    def parse_date(s: str):
        if '-' in s:
            y, m, d = s.split('-')
            return (int(y), int(m), int(d))
        else:
            d, m, y = s.split('.')
            return (int(y), int(m), int(d))
    a = parse_date(payload['a'])
    b = parse_date(payload['b'])
    if a < b:
        label = 'before'
    elif a > b:
        label = 'after'
    else:
        label = 'same'
    return {'label': label}
