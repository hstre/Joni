def solve(payload: dict) -> dict:
    import re
    from decimal import Decimal

    # conversion factors to meters (base)
    factors = {
        'km': Decimal(1000),
        'm': Decimal(1),
        'cm': Decimal('0.01'),
        'mm': Decimal('0.001'),
        'mile': Decimal('1609.344'),
        'mi': Decimal('1609.344'),
        'foot': Decimal('0.3048'),
        'ft': Decimal('0.3048'),
        'inch': Decimal('0.0254'),
        'in': Decimal('0.0254')
    }

    def parse(s: str):
        s = s.strip()
        m = re.fullmatch(r'([+-]?\d+(?:\.\d+)?)\s*([a-zA-Z]+)', s)
        if not m:
            raise ValueError(f"Invalid measurement: {s}")
        num = Decimal(m.group(1))
        unit = m.group(2).lower()
        if unit not in factors:
            raise ValueError(f"Unknown unit: {unit}")
        return num * factors[unit]

    try:
        val_a = parse(payload['a'])
        val_b = parse(payload['b'])
        if val_a == val_b:
            return {'label': 'same'}
        else:
            return {'label': 'different'}
    except Exception:
        return {'label': 'different'}
