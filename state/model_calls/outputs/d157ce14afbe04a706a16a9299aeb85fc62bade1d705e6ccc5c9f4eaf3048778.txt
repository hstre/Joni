def solve(payload: dict) -> dict:
    import re

    units = {
        'mm': 0.001,
        'cm': 0.01,
        'm': 1.0,
        'km': 1000.0,
        'inch': 0.0254,
        'in': 0.0254,
        'ft': 0.3048,
        'foot': 0.3048,
        'feet': 0.3048,
        'yd': 0.9144,
        'yard': 0.9144,
        'mi': 1609.344,
        'mile': 1609.344,
        'miles': 1609.344,
        'kilometer': 1000.0,
        'kilometers': 1000.0,
        'centimeter': 0.01,
        'centimeters': 0.01,
        'millimeter': 0.001,
        'millimeters': 0.001,
        'meter': 1.0,
        'meters': 1.0,
    }

    def parse(s: str) -> float:
        s = s.strip()
        s = re.sub(r'(\d),(\d)', r'\1\2', s)
        m = re.match(r'^([-+]?\d*\.?\d+)\s*([a-zA-Z]+)$', s)
        if not m:
            raise ValueError
        num_str, unit_str = m.groups()
        factor = units.get(unit_str.lower().strip())
        if factor is None:
            if unit_str.lower().endswith('s'):
                factor = units.get(unit_str.lower().strip()[:-1])
            if factor is None:
                raise ValueError
        return float(num_str) * factor

    try:
        v1 = parse(payload['a'])
        v2 = parse(payload['b'])
    except (KeyError, ValueError):
        return {'label': 'different'}

    tol = 1e-9 * max(abs(v1), abs(v2), 1.0)
    return {'label': 'same'} if abs(v1 - v2) <= tol else {'label': 'different'}
